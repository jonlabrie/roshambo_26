# Roblox Toolchain + Walking Skeleton Implementation Plan (Milestone 2)

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** A `roblox/` project with professional toolchain (Rokit/Rojo/Lune/StyLua/Selene), TDD'd core modules (GameRules mirror, RoundClock, NetworkClient, RoundCoordinator), and a Studio place that connects to the live `/api/v1` and prints round flow to the Output window.

**Architecture:** Modules are **dependency-injected and never `require` each other** — the same `.luau` files load under Lune (path requires, headless tests, CI) and Roblox (instance requires via Rojo mapping). The only Roblox-runtime file is the composition root `main.server.luau`. Spec: `docs/superpowers/specs/2026-06-05-roblox-client-design.md` §7/§8, milestone 2 of §11.

**Tech Stack:** Luau, Rokit (toolchain pinning), Rojo 7 (file↔Studio sync), Lune (headless Luau runtime for tests), StyLua + Selene (format/lint), GitHub Actions. **Deviation from spec §7:** Jest-Lua requires the Roblox runtime and cannot run under Lune/CI, so tests use a minimal bespoke harness (`tests/harness.luau`, jest-style API); Wally is deferred until the first real package dependency exists (YAGNI).

**Conventions:** All commands run from `roblox/` unless noted. Run tests: `lune run tests/run`. The backend API contract is implemented in `server/src/routes/apiV1.ts` and live at `https://fiuuwhrqgi.us-east-1.awsapprunner.com`; local stack: `docker compose up -d mongodb` + `cd server && npm run dev` (needs `MONGODB_URI=mongodb://localhost:27017/roshambo`, `API_KEY=roshambo_local_dev_api_key`, `TEST_MODE=true`).

```lua
--!strict
-- Minimal jest-style test harness for Lune. Replace with Jest-Lua if it ever
-- gains Lune support (today it requires the Roblox runtime).
local harness = {}

type Test = { name: string, fn: () -> () }
local tests: { Test } = {}
local prefix = ""

function harness.describe(name: string, fn: () -> ())
    local prev = prefix
    prefix = prefix .. name .. " > "
    fn()
    prefix = prev
end

function harness.test(name: string, fn: () -> ())
    table.insert(tests, { name = prefix .. name, fn = fn })
end

local function deepEqual(a: any, b: any): boolean
    if a == b then
        return true
    end
    if type(a) ~= "table" or type(b) ~= "table" then
        return false
    end
    for k, v in a :: { [any]: any } do
        if not deepEqual(v, (b :: { [any]: any })[k]) then
            return false
        end
    end
    for k in b :: { [any]: any } do
        if (a :: { [any]: any })[k] == nil then
            return false
        end
    end
    return true
end

local function fmt(v: any): string
    if type(v) == "table" then
        local parts = {}
        for k, val in v :: { [any]: any } do
            table.insert(parts, tostring(k) .. "=" .. fmt(val))
        end
        table.sort(parts)
        return "{" .. table.concat(parts, ", ") .. "}"
    end
    return tostring(v)
end

function harness.expect(actual: any)
    return {
        toBe = function(expected: any)
            if actual ~= expected then
                error(("expected %s to be %s"):format(fmt(actual), fmt(expected)), 2)
            end
        end,
        toEqual = function(expected: any)
            if not deepEqual(actual, expected) then
                error(("expected %s to deep-equal %s"):format(fmt(actual), fmt(expected)), 2)
            end
        end,
        toBeCloseTo = function(expected: number, eps: number?)
            local e = eps or 1e-6
            if math.abs((actual :: number) - expected) > e then
                error(("expected %s to be within %s of %s"):format(fmt(actual), tostring(e), fmt(expected)), 2)
            end
        end,
        toBeTruthy = function()
            if not actual then
                error(("expected %s to be truthy"):format(fmt(actual)), 2)
            end
        end,
        toBeNil = function()
            if actual ~= nil then
                error(("expected %s to be nil"):format(fmt(actual)), 2)
            end
        end,
        toThrow = function()
            local ok = pcall(actual :: () -> ())
            if ok then
                error("expected function to throw", 2)
            end
        end,
    }
end

function harness.run(): boolean
    local passed, failed = 0, 0
    for _, t in tests do
        local ok, err = pcall(t.fn)
        if ok then
            passed += 1
        else
            failed += 1
            print(("FAIL  %s\n      %s"):format(t.name, tostring(err)))
        end
    end
    print(("\n%d passed, %d failed, %d total"):format(passed, failed, passed + failed))
    return failed == 0
end

-- Allows the runner to detect double-loading mistakes in specs.
function harness.count(): number
    return #tests
end

return harness
```

--!strict
-- Luau mirror of server/src/engine/GameRules.ts. The shared JSON fixtures are
-- the contract; if you change one side, the other side's CI fails.
local GameRules = {}

export type Throw = "R" | "P" | "S"
export type RoundResult = "WIN" | "SAFE" | "LOSS"

local BEATS: { [string]: string } = { R = "S", P = "R", S = "P" }

function GameRules.calculateResult(player: Throw, world: Throw): RoundResult
    if player == world then
        return "SAFE"
    end
    return if BEATS[player] == world then "WIN" else "LOSS"
end

-- Pot math: 0 -> 1 -> 3 -> 9 -> 27 ... (3^n). SAFE preserves the pot; LOSS forfeits it.
function GameRules.nextPot(currentPot: number, result: RoundResult): number
    if result == "WIN" then
        return if currentPot == 0 then 1 else currentPot * 3
    end
    if result == "LOSS" then
        return 0
    end
    return currentPot
end

-- On WIN the delta is the NEW pot value (not the increment); LOSS is -old pot; SAFE is 0.
function GameRules.potDelta(currentPot: number, result: RoundResult): number
    if result == "WIN" then
        return GameRules.nextPot(currentPot, "WIN")
    end
    if result == "LOSS" then
        return if currentPot == 0 then 0 else -currentPot
    end
    return 0
end

-- Streaks survive only on WIN.
function GameRules.nextStreak(currentStreak: number, result: RoundResult): number
    return if result == "WIN" then currentStreak + 1 else 0
end

return GameRules
```

    end)

        local c = RoundClock.new()

        local c = RoundClock.new()
        c:addSample(1000, 100, 600) -- offset 450

    end)

--!strict
-- Estimates the offset between local clock and the game server's clock from
-- /state samples. Keeps the lowest-RTT sample (Cristian's algorithm): a
-- routing spike inflates RTT, so spiked samples are discarded rather than
-- averaged in. Design tolerance is seconds (spec §4), so RTT/2 asymmetry
-- error is immaterial.
local RoundClock = {}
RoundClock.__index = RoundClock

export type RoundClock = typeof(setmetatable(
    {} :: { _offsetMs: number?, _bestRttMs: number },
    RoundClock
))

function RoundClock.new(): RoundClock
    return setmetatable({ _offsetMs = nil :: number?, _bestRttMs = math.huge }, RoundClock)
end

function RoundClock.addSample(self: RoundClock, serverTimeMs: number, rttMs: number, localReceiveMs: number)
    if rttMs < self._bestRttMs then
        self._bestRttMs = rttMs
        self._offsetMs = (serverTimeMs + rttMs / 2) - localReceiveMs
    end
end

function RoundClock.hasSync(self: RoundClock): boolean
    return self._offsetMs ~= nil
end

function RoundClock.toServerTime(self: RoundClock, localMs: number): number
    assert(self._offsetMs, "RoundClock has no sync yet")
    return localMs + self._offsetMs
end

function RoundClock.toLocalTime(self: RoundClock, serverMs: number): number
    assert(self._offsetMs, "RoundClock has no sync yet")
    return serverMs - self._offsetMs
end

function RoundClock.reset(self: RoundClock)
    self._offsetMs = nil
    self._bestRttMs = math.huge
end

return RoundClock
```

-- HTTP client for the Roshambo /api/v1 (server-to-server). All side effects
-- (transport, json, sleep, clock, random) are injected so this module runs
-- under Lune tests and Roblox alike. Never reaches Roblox clients.
local NetworkClient = {}
NetworkClient.__index = NetworkClient

export type Response = { ok: boolean, statusCode: number?, body: string? }
export type Deps = {
    request: (req: { Url: string, Method: string, Headers: { [string]: string }, Body: string? }) -> Response,
    jsonDecode: (s: string) -> any,
    sleep: (seconds: number) -> (),
    now: () -> number, -- local epoch ms
    random: () -> number, -- [0,1)
}
export type Config = { baseUrl: string, apiKey: string, maxAttempts: number? }

export type Result = {
    ok: boolean,
    data: any?,
    status: number?,
    error: string?,
    notReady: boolean?,
    rttMs: number?,
    localReceiveMs: number?,
}

function NetworkClient.new(config: Config, deps: Deps)
    return setmetatable({
        _base = config.baseUrl,
        _key = config.apiKey,
        _maxAttempts = config.maxAttempts or 3,
        _deps = deps,
    }, NetworkClient)
end

local function backoffSeconds(attempt: number, random: () -> number): number
    -- 0.5 * 2^(attempt-1) plus up to 0.5s jitter
    return 0.5 * 2 ^ (attempt - 1) + random() * 0.5
end

function NetworkClient._get(self: any, path: string): Result
    local d: Deps = self._deps
    for attempt = 1, self._maxAttempts do
        local sentAt = d.now()
        local resp = d.request({
            Url = self._base .. path,
            Method = "GET",
            Headers = { ["X-API-Key"] = self._key },
        })
        local receivedAt = d.now()
        if resp.ok and resp.statusCode then
            local code = resp.statusCode
            if code >= 200 and code < 300 then
                return {
                    ok = true,
                    data = d.jsonDecode(resp.body or "null"),
                    status = code,
                    rttMs = receivedAt - sentAt,
                    localReceiveMs = receivedAt,
                }
            end
            if code == 404 then
                return { ok = false, notReady = true, status = code }
            end
            if code < 500 then
                -- 4xx other than 404: our bug or config problem; retrying won't help
                return { ok = false, status = code, error = "HTTP_" .. tostring(code) }
            end
            -- 5xx falls through to retry
        end
        if attempt < self._maxAttempts then
            d.sleep(backoffSeconds(attempt, d.random))
        end
    end
    return { ok = false, error = "TRANSPORT_FAILED" }
end

function NetworkClient.getState(self: any): Result
    return self:_get("/api/v1/state")
end

function NetworkClient.getRoundResult(self: any, roundId: string): Result
    return self:_get(`/api/v1/rounds/{roundId}/result`)
end

return NetworkClient
```

--!strict
-- Walking-skeleton coordinator: polls /state ~1Hz, feeds the RoundClock,
-- logs round/phase transitions, and fetches each round's result exactly once
-- when it enters TALLY/REVEAL. Throw submission arrives in milestone 3.
local RoundCoordinator = {}
RoundCoordinator.__index = RoundCoordinator

export type Deps = {
    net: any, -- NetworkClient-shaped: getState, getRoundResult
    clock: any, -- RoundClock-shaped: addSample
    log: (msg: string) -> (),
    random: () -> number, -- [0,1) for poll jitter
}

function RoundCoordinator.new(deps: Deps)
    return setmetatable({
        _deps = deps,
        _roundId = nil :: string?,
        _phase = nil :: string?,
        _roundCount = 0,
        _resultLogged = false,
    }, RoundCoordinator)
end

-- Returns the suggested delay (seconds) until the next poll.
function RoundCoordinator.pollOnce(self: any): number
    local d: Deps = self._deps
    local res = d.net:getState()
    if not res.ok then
        d.log(`[NET] state failed: {res.error or res.status}`)
        return 3 -- back off harder when the server is unreachable
    end

    local state = res.data
    d.clock:addSample(state.serverTime, res.rttMs, res.localReceiveMs)

    if state.roundId ~= self._roundId then
        self._roundId = state.roundId
        self._roundCount = state.roundCount
        self._phase = nil
        self._resultLogged = false
    end

    if state.phase ~= self._phase then
        self._phase = state.phase
        d.log(`[ROUND {self._roundCount}] {self._roundId} {state.phase}`)
    end

    if (state.phase == "TALLY" or state.phase == "REVEAL") and not self._resultLogged then
        local result = d.net:getRoundResult(self._roundId)
        if result.ok then
            self._resultLogged = true
            local r = result.data
            d.log(
                `[ROUND {self._roundCount}] WORLD THROW: {r.worldThrow}`
                    .. ` | R {r.distribution.R}% P {r.distribution.P}% S {r.distribution.S}%`
                    .. ` | players {r.totalPlayers}`
            )
        end
        -- notReady: leave the latch open; next poll retries
    end

    return 1 + d.random() * 0.25
end

return RoundCoordinator
```

-- Composition root: wires Roblox services into the runtime-agnostic modules.
local HttpService = game:GetService("HttpService")
local ReplicatedStorage = game:GetService("ReplicatedStorage")

local NetworkClient = require(script.Parent:WaitForChild("NetworkClient"))
local RoundCoordinator = require(script.Parent:WaitForChild("RoundCoordinator"))

local RoundClock = require(shared:WaitForChild("RoundClock"))

local secretsModule = script.Parent:FindFirstChild("SecretsLocal") or script.Parent:WaitForChild("SecretsExample")
local config = require(secretsModule) :: { baseUrl: string, apiKey: string }

local deps = {
    request = function(req)
        local ok, resp = pcall(function()
            return HttpService:RequestAsync({
                Url = req.Url,
                Method = req.Method,
                Headers = req.Headers,
                Body = req.Body,
            })
        end)
        if not ok then
            return { ok = false }
        end
        return { ok = true, statusCode = resp.StatusCode, body = resp.Body }
    end,
    jsonDecode = function(s: string)
        return HttpService:JSONDecode(s)
    end,
    sleep = function(seconds: number)
        task.wait(seconds)
    end,
    now = function()
        return DateTime.now().UnixTimestampMillis
    end,
    random = function()
        return math.random()
    end,
}

local net = NetworkClient.new({ baseUrl = config.baseUrl, apiKey = config.apiKey }, deps)
local clock = RoundClock.new()
local coordinator = RoundCoordinator.new({
    net = net,
    clock = clock,
    log = print,
    random = math.random,
})

print(`[ROSHAMBO] walking skeleton starting against {config.baseUrl}`)
task.spawn(function()
    while true do
        local delay = coordinator:pollOnce()
        task.wait(delay)
    end
end)
